select * from COURSES;
select * from APPLIED_COURSES;

drop table LOGINS;
drop table COURSES;
drop table APPLIED_COURSES



CREATE TABLE LOGINS(
    STUDENT_ID INT PRIMARY KEY AUTO_INCREMENT,
    USERID VARCHAR(50) NOT NULL,
    PASSWORD VARCHAR(50) NOT NULL,
    NAME VARCHAR(50) NOT NULL,
    MAJOR VARCHAR(50) NOT NULL,
    GRADE INT NOT NULL
);

CREATE TABLE APPLIED_COURSES (
    NUMBER          CHARACTER VARYING(10)  NOT NULL,
    A_DIVISION      CHARACTER VARYING(10)  NOT NULL,
    A_CREDITS       INT,
    A_TIME          CHARACTER VARYING(10)  NOT NULL,
    A_ID            CHARACTER VARYING(10)  NOT NULL,
    A_NAME          CHARACTER VARYING(50)  NOT NULL,
    A_PROFESSOR     CHARACTER VARYING(20),
    A_RATING        CHARACTER VARYING(10)
);

CREATE TABLE COURSES (
    ID VARCHAR(10) PRIMARY KEY,
    DIVISION VARCHAR(20),
    CREDITS INT,
    TIME CHARACTER VARYING(10)  NOT NULL,
    NAME VARCHAR(50),
    PROFESSOR VARCHAR(20),
    RATING VARCHAR(10)
);
